# Checks for tools/boot-test.py on the live ISO with a blank disk attached
# (BOOT_DISK=<outdir>/disk.raw, 24 GB): lay the disk out like a Windows PC,
# install bunix into its free space (a console install), and check that
# nothing of "Windows" changed.  Then boot the disk with
# tools/boot-checks/dual-boot.txt.
#
# The "Windows" disk: a 100 MiB ESP with EFI/Microsoft/Boot/bootmgfw.efi
# (a stand-in: a copy of systemd-boot), MSR, C: and the recovery partition
# (random data, checksummed), free space between C: and recovery, and a
# "Windows Boot Manager" firmware entry.  The root shell is fish.

systemctl is-system-running; lsblk -o NAME,SIZE,TYPE,LABEL
printf 'label: gpt\nsize=100MiB, type=C12A7328-F81F-11D2-BA4B-00A0C93EC93B, name="EFI system partition"\nsize=16MiB, type=E3C9E316-0B5C-4DB8-817D-F92DF00215AE, name="Microsoft reserved partition"\nsize=4GiB, type=EBD0A0A2-B9E5-4433-87C0-68B6B72699C7, name="Basic data partition"\nstart=20GiB, size=600MiB, type=DE94BBA4-06D1-4D40-A16A-BFD50179D6AC, name="Windows recovery"\n' | sfdisk -q /dev/vda; sfdisk -d /dev/vda
mkfs.vfat -F 32 -n SYSTEM /dev/vda1 >/dev/null; mkdir -p /mnt/esp /mnt/cd; mount /dev/vda1 /mnt/esp; mount -o ro /dev/sr0 /mnt/cd; mkdir -p /mnt/esp/EFI/Microsoft/Boot; cp /mnt/cd/install/esp/EFI/bunix/systemd-bootx64.efi /mnt/esp/EFI/Microsoft/Boot/bootmgfw.efi; umount /mnt/esp /mnt/cd; echo esp ready
for p in vda2 vda3 vda4; dd if=/dev/urandom of=/dev/$p bs=1M count=16 status=none; end; for p in vda2 vda3 vda4; head -c 16M /dev/$p | sha256sum; end > /root/before.sha256; cat /root/before.sha256
efibootmgr -q -c -d /dev/vda -p 1 -L 'Windows Boot Manager' -l '\EFI\Microsoft\Boot\bootmgfw.efi'; efibootmgr
# the menus, without answering the second question
printf '1\n' | bunix-install; echo "exit $status"
# a console install (2), for the user "tester"
printf '1\n4\n2\ntest-pc\nTest User\ntester\nsecret\nsecret\nyes\n' | bunix-install; echo "install exit $status"
# the accounts: tester in wheel, root's password locked; then root's empty
# password again, so dual-boot.txt gets console-shell's root shell
mkdir -p /mnt/root; mount /dev/disk/by-partlabel/root /mnt/root; grep -E '^(root|tester):' /mnt/root/etc/passwd; grep '^wheel:' /mnt/root/etc/group; awk -F: '$1 == "root" || $1 == "tester" { print $1, substr($2, 1, 4) }' /mnt/root/etc/shadow; ls /mnt/root/var/lib/extensions 2>&1; usermod --prefix /mnt/root --password '' root; umount /mnt/root
sfdisk -d /dev/vda; lsblk -o NAME,SIZE,PARTTYPENAME,PARTLABEL,FSTYPE /dev/vda
for p in vda2 vda3 vda4; head -c 16M /dev/$p | sha256sum; end > /root/after.sha256; sha256sum /root/before.sha256 /root/after.sha256
mount /dev/vda1 /mnt/esp; find /mnt/esp -type f; umount /mnt/esp; efibootmgr
# the next boot takes the bunix entry: the stand-in "Windows" would start systemd-boot itself
efibootmgr -q -n (efibootmgr | string match -r '^Boot([0-9A-F]{4})\* bunix')[2]; efibootmgr | grep -E 'BootNext|BootOrder'; sync; echo synced
