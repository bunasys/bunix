# Checks for tools/boot-test.py (BOOT_CHECKS=...): phase 4 -- PAM, logind
# sessions, the per-user manager, users and polkit (docs/desktop.md).
# `runuser -l` goes through PAM's login rules (pam_systemd) like the
# console's login does.  Root's shell is fish.

systemctl is-system-running; systemctl --failed --no-legend --no-pager; test -e /run/nologin; and echo "nologin still there"; or echo "logins open"
useradd -m -G wheel -c 'Test User' test; and echo 'test:bunix-test' | chpasswd; getent passwd test; id test; ls -ld /home/test
# new passwords are yescrypt ($y$), through bunix's PAM rules: nothing in
# /etc/pam.d overrides /usr/lib/pam.d
string match -q 'test:$y$*' (grep '^test:' /etc/shadow); and echo "hash yescrypt"; or echo "HASH NOT YESCRYPT"; ls /etc/pam.d 2>&1; systemctl is-active bunix-pam-cleanup.service
runuser -l test -c 'id -un; echo "runtime $XDG_RUNTIME_DIR"; echo "session $XDG_SESSION_ID"'
sleep 2; loginctl list-users --no-legend --no-pager; systemctl is-active user@1000.service
runuser -l test -c 'systemctl --user is-system-running'
find /usr -perm -4000 -type f; ls -l /usr/bin/passwd
pkaction | wc -l; systemctl is-active polkit.service polkit-agent-helper.socket
runuser -l test -c 'pkcheck --action-id org.freedesktop.systemd1.manage-units --process $fish_pid; echo "pkcheck exit $status"'
# systemd built with polkit: a wheel user is asked to authenticate
# ("Interactive authentication required"), not refused ("Access denied")
runuser -l test -c 'run0 --no-ask-password true 2>&1; echo "run0 as test exit $status"'
run0 id -un; journalctl -b -p err --no-pager -o cat | head -8
